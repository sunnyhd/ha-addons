#include <tunables/global>

# The name below is a placeholder: the Supervisor rewrites it to the installed
# slug at load time, and the slug carries an install-dependent prefix (`local_`
# for a local add-on, the repository hash for a store one). Nothing in this file
# may therefore reference the profile by name -- which is why the child receive
# rules are peerless.
#
# Adapted from the base profile of local-audio-addon. Two things differ here: the
# bundled dbus and avahi daemons are disabled in this add-on (the image removes
# them from the s6 user chain), so their child profiles are absent; and one
# sendspin-cli runs per zone instead of a single one -- all under the same child
# profile a single instance would get.
profile zone_players flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # A fixed-function add-on: no mount/umount/remount, no dac_read_search, no
  # sys_admin. The supervision tree, bashio and the option-reading scripts all
  # run as root and switch to no other user, so CAP_KILL is the only capability
  # this outer profile declares. s6-supervise signals the longrun down -- the
  # SIGTERM and the SIGKILL its timeout-kill fires behind it -- and both are a
  # CAP_KILL check.
  capability kill,

  file,

  # s6 supervises the service and passes on a container stop. The stop signal
  # arrives from the unconfined container runtime, which the base abstraction
  # already permits receiving from. The child profile below repeats this set as
  # its `signal (receive)` -- they have to match, or a signal is denied at
  # delivery.
  signal (send) set=(kill,term,int,hup,cont,usr1,usr2),

  # s6-overlay v3: the init, the execline and s6 tools it puts on PATH, and the
  # supervision state it builds under /run.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /command/** ix,
  /package/** ix,
  /etc/s6-overlay/** r,
  /run/{s6,s6-rc*,service}/** rwix,
  /run/s6-linux-init-container-results/** rw,

  # bashio reads the options over the Supervisor API; this add-on's shared shell
  # library (common.sh) and jq turn the answer into the values the run script
  # writes into the player configs.
  /usr/lib/bashio/** ix,
  /usr/lib/zone-players/** r,

  # bashio's option fetch is an HTTP call to the Supervisor; resolving the
  # hostname for it needs the dgram pair. pactl reaches PulseAudio over a unix
  # socket, the players reach the Music Assistant server over inet.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,

  # /run/zone-players holds the rendered configs and the control sockets per
  # zone; /data is the persistent volume (instance id and the per-zone state
  # under /data/zones); s6 writes and locks the rest of /run. pactl talks to the
  # mapped-in PulseAudio under /run/audio -- covered by the blanket `file,` rule
  # above.
  /run/zone-players/** rw,
  /data/** rw,
  /run/** rwk,
  /tmp/** rw,

  # Nothing here reads the shadow files or edits the account databases, and the
  # `file` rule above would otherwise allow both.
  deny /etc/shadow rw,
  deny /etc/gshadow rw,
  deny /etc/{passwd,group} w,

  # The player is confined more tightly than this profile.
  /usr/bin/sendspin-cli cx,

  # One sendspin-cli per zone, all under this single child profile. It needs no
  # capability: it runs as root, and what it opens is root-owned.
  profile /usr/bin/sendspin-cli flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    signal (receive) set=(kill,term,int,hup,cont,usr1,usr2),

    /usr/bin/sendspin-cli mr,

    # A wss:// server value needs the trust store.
    /etc/ssl/** r,
    /usr/share/ca-certificates/** r,

    # The PulseAudio client writes its cookie under the (root) home.
    owner /root/.config/ w,
    owner /root/.config/pulse/ w,
    owner /root/.config/pulse/** rw,

    # Its own rendered config, control socket and persistent state per zone. k for
    # the socket's lock file.
    /run/zone-players/** rwk,
    /data/zones/** rw,

    # ALSA: the library's configuration and the plugin that bridges to
    # PulseAudio.
    /usr/lib/**/alsa-lib/*.so mr,
    /usr/share/alsa/** r,
    /etc/alsa/** r,
    /etc/asound.conf r,
    /dev/snd/ r,
    /dev/snd/** rw,
    @{PROC}/asound/** r,
    /sys/class/sound/** r,
    /sys/devices/**/sound/** r,

    # Home Assistant's PulseAudio, which `audio: true` maps in and through which
    # each player plays its `pulse:<sink>` output.
    /etc/pulse/** r,
    /run/audio/** rw,
    /dev/shm/** rw,

    # The websocket server (port 8928 upward, one per zone) and the connections
    # it makes out; UDP to resolve the `server` value.
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,
    network unix dgram,
  }
}
